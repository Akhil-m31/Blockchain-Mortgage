pragma solidity ^0.8.0;

contract Mortgage {

    struct Loan {
        address borrower;
        address payable lender;
        uint256 amount;
        uint256 interest;
        uint256 duration;
        bool approved;
        uint256 emiPaid;
        uint256 totalEmi;
        string documentHash;
        bool closed;
    }

    event LoanApplied(address indexed borrower, uint256 amount, uint256 interest, uint256 duration);
    event LoanApproved(address indexed borrower, address indexed lender, uint256 amount);
    event EMIPaid(address indexed borrower, address indexed lender, uint256 amount, uint256 emiIndex);
    event LoanClosed(address indexed borrower);

    mapping(address => Loan) public loans;

    // ─── Borrower applies for a loan ───────────────────────────────────────────
    function applyLoan(uint256 amount, uint256 interest, uint256 duration) public {
        loans[msg.sender] = Loan(
            msg.sender,
            payable(address(0)),
            amount,
            interest,
            duration,
            false,
            0,
            duration,
            "",
            false
        );
        emit LoanApplied(msg.sender, amount, interest, duration);
    }

    // ─── Read loan details ─────────────────────────────────────────────────────
    function getLoan(address user) public view returns (
        address,
        uint256,
        uint256,
        uint256,
        bool,
        uint256,
        uint256,
        string memory,
        bool
    ) {
        Loan memory loan = loans[user];
        return (
            loan.borrower,
            loan.amount,
            loan.interest,
            loan.duration,
            loan.approved,
            loan.emiPaid,
            loan.totalEmi,
            loan.documentHash,
            loan.closed
        );
    }

    // ─── Get lender address ────────────────────────────────────────────────────
    function getLender(address borrower) public view returns (address) {
        return loans[borrower].lender;
    }

    // ─── Lender approves AND funds the loan ───────────────────────────────────
    function approveLoan(address payable borrower) public payable {
        require(loans[borrower].amount > 0, "Loan does not exist");
        require(!loans[borrower].approved,  "Loan already approved");
        require(msg.value == loans[borrower].amount, "Must send exact loan amount in ETH");

        loans[borrower].approved = true;
        loans[borrower].lender   = payable(msg.sender);

        // Modern safe ETH transfer to borrower
        (bool sent, ) = borrower.call{value: msg.value}("");
        require(sent, "Failed to send ETH to borrower");

        emit LoanApproved(borrower, msg.sender, msg.value);
    }

    // ─── Calculate EMI amount ──────────────────────────────────────────────────
    function calculateEMI(address borrower) public view returns (uint256) {
        Loan memory loan = loans[borrower];
        require(loan.approved, "Loan not approved");

        uint256 totalAmount = loan.amount + (loan.amount * loan.interest / 100);
        return totalAmount / loan.duration;
    }

    // ─── Borrower pays one EMI installment ────────────────────────────────────
    function payEMI() public payable {
        Loan storage loan = loans[msg.sender];
        require(loan.approved,                "Loan not approved");
        require(!loan.closed,                 "Loan already closed");
        require(loan.emiPaid < loan.totalEmi, "All EMIs already paid");

        uint256 expectedEmi = calculateEMI(msg.sender);
        require(msg.value == expectedEmi, "Must send exact EMI amount in ETH");

        loan.emiPaid += 1;

        // Modern safe ETH transfer to lender
        (bool sent, ) = loan.lender.call{value: msg.value}("");
        require(sent, "Failed to send ETH to lender");

        emit EMIPaid(msg.sender, loan.lender, msg.value, loan.emiPaid);
    }

    // ─── Store IPFS document hash ──────────────────────────────────────────────
    function setDocumentHash(string memory hash) public {
        require(loans[msg.sender].amount > 0, "Loan does not exist");
        loans[msg.sender].documentHash = hash;
    }

    // ─── Close the loan once all EMIs are paid ────────────────────────────────
    function closeLoan() public {
        Loan storage loan = loans[msg.sender];
        require(loan.emiPaid == loan.totalEmi, "EMIs not fully paid");
        loan.closed = true;
        emit LoanClosed(msg.sender);
    }
}
